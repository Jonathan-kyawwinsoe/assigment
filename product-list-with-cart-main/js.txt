const addBtn = document.getElementById("add_to_card");
const quanity = document.getElementById("quantity");
const count_quanity = document.getElementById("count");
const minusBtn = document.getElementById("minus-btn");
const plusBtn = document.getElementById("plus-btn");

let count = 1;

addBtn.addEventListener("click", () => {
  addBtn.classList.add("hidden");
  quanity.classList.remove("hidden");
  quanity.classList.add("flex");
  count = 1;
  count_quanity.innerText = count;
});

plusBtn.addEventListener("click", () => {
  count++;
  count_quanity.innerHTML = count;
});

minusBtn.addEventListener("click", () => {
  if (count > 1) {
    count--;
    count_quanity.innerText = count;
  } else {
    quanity.classList.add("hidden");
    quanity.classList.remove("flex");
    addBtn.classList.remove("hidden");
  }
});
